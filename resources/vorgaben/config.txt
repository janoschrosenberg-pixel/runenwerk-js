# ---------------------------------------------------------------- Arbeitsbereiche
# TypeScript/React-Projekt und Java-Projekt; beide werden indiziert (Strg+P sucht in beiden).
# Ohne ts.workspace fragt der Editor beim Start danach (und traegt ihn hier ein).

java.workspace =

# ---------------------------------------------------------------- Language Server
# TypeScript: npm install -g typescript typescript-language-server (node_modules/.bin geht vor)
ts.lsp = typescript-language-server --stdio
# Java: Eclipse jdtls ist in runenjs.jar eingebaut und wird beim ersten Start daneben entpackt (jdtls-eingebaut).
# Nur fuer ein eigenes jdtls: Ordner angeben (oder java.lsp = Befehl im PATH); java.lsp = off schaltet ihn ab
java.lsp.home =
java.lsp = jdtls
# Java 21+ fuer jdtls (leer: wird gesucht - das laufende Java, JAVA_HOME, uebliche JDK-Ordner) und sein Speicher
java.lsp.java =
java.lsp.memory = 2G

# Java mit einem Eclipse-Formatter-Profil (XML-Export aus Eclipse oder IntelliJ "Eclipse XML") ...

# ... oder mit einem IntelliJ-Code-Style (XML) ueber den IntelliJ-Formatter (IntelliJ muss dafuer zu sein)

java.format.intellij.cmd =
